# VisManager runtime + build dependencies
#
# Pillow    - image decoding for TGA/PNG/JPEG/BMP/GIF/WebP/TIFF/ICO/DDS
# pypdf     - PDF preview via embedded-image extraction (PURE PYTHON,
#             survives PyInstaller packaging with no extra configuration)
# pyinstaller - builds the executable
#
# tkinter is part of the Python standard library. On Windows and macOS it
# ships with the official python.org installer. On Debian/Ubuntu it is a
# separate package:  sudo apt install python3-tk

Pillow>=10.0.0
pypdf>=4.0.0
pyinstaller>=6.0.0

# tkinterdnd2 enables dragging files and folders onto the window. Without it
# the app works normally and files are opened through the dialog.
tkinterdnd2>=0.4.0

# ── Optional: full PDF renderer ───────────────────────────────────────────
# pypdfium2 rasterises any PDF, including vector and text-only documents.
# pypdf alone can only pull out embedded images, which covers every PDF
# VisManager produces plus virtually all scanned documents.
#
# It ships a native binary that PyInstaller does not detect on its own —
# vismanager.spec collects it explicitly. Uncomment to enable:
#
# pypdfium2>=4.0.0

# ── Optional: 3D cube file viewer ─────────────────────────────────────────
# Adds interactive .cube support (isosurfaces, rotate/zoom, and export to
# PNG/TIFF/JPEG at 1x-8x, or SVG/PDF/EPS as true vector).
#
# VTK is ~500 MB installed and will enlarge a packaged build considerably.
# VisManager imports it optionally: without it everything else works exactly
# the same, minus .cube files. Uncomment to enable:
#
# vtk>=9.2

# ── Optional: generate cube files from quantum-chemistry output ───────────
# pyscf converts molden files into HOMO/LUMO cubes. The .gbw and density
# stages additionally need ORCA's orca_2mkl and orca_plot on PATH; those
# ship with ORCA and cannot be pip-installed.
#
# pyscf>=2.3
